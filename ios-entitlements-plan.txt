# DoseRoutine — TestFlight Submission Entitlements Plan
#
# Source of truth for what capabilities the iOS app is ALLOWED to declare.
# The check-entitlements script diffs the built .ipa's embedded entitlements
# and the ASC capabilities list against this file and fails on any drift.
#
# Format: one key per line. Comments start with #. Sections:
#   [required]   MUST be present in the built app
#   [optional]   MAY be present; not flagged either way
#   [forbidden]  MUST NOT be present (would trigger extra App Review scrutiny)
#
# When you intentionally add or remove a capability, update this file in the
# same commit as the Xcode/Capacitor change. That commit becomes the audit trail.

[required]
application-identifier
com.apple.developer.team-identifier
# HealthKit: the Fitness/Body screens read weight, steps, active energy and
# heart rate through @capgo/capacitor-health (src/lib/health-bridge.ts) and the
# app degrades to manual entry when the user denies access. Usage strings live
# in ios/App/App/Info.plist.
com.apple.developer.healthkit
# Universal links and web credentials for doseroutine.com.
com.apple.developer.associated-domains
# Sign in with Apple: offered on /auth (src/routes/auth.tsx) and required by
# guideline 4.8 because the web build also offers Google sign-in.
com.apple.developer.applesignin

[optional]
# StoreKit / IAP does NOT require an explicit entitlement — Apple grants it
# implicitly when the app has active in-app products. RevenueCat uses StoreKit
# under the hood, so no entitlement key is needed. `com.apple.developer.in-app-payments`
# is the Apple Pay (merchant IDs) entitlement, not StoreKit IAP — do not add it
# unless the app actually integrates Apple Pay.
com.apple.developer.in-app-payments
# Push notifications: enabled for dose reminders only if the native build
# opts in. Absent in the current Capacitor build; harmless if added later.
aps-environment
# Background modes for local notification scheduling
com.apple.developer.background-modes
# Keychain sharing group used by Capacitor Preferences
keychain-access-groups
# get-task-allow is TRUE for debug builds, FALSE/absent for release
get-task-allow
